cmake_minimum_required(VERSION 3.20)
project(LifeEngineNoodleBootstrap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(noodle_frontend src/lexer.cpp src/parser.cpp src/diagnostics.cpp src/codegen.cpp)
target_include_directories(noodle_frontend PUBLIC include)
add_library(noodle_runtime src/interpreter.cpp)
target_include_directories(noodle_runtime PUBLIC include)
target_link_libraries(noodle_runtime PUBLIC noodle_frontend)
add_executable(noodle src/main.cpp)
target_link_libraries(noodle PRIVATE noodle_runtime)
add_executable(nudlc src/main.cpp)
target_link_libraries(nudlc PRIVATE noodle_runtime)
enable_testing()
add_executable(noodle_smoke_test tests/smoke_test.cpp)
target_link_libraries(noodle_smoke_test PRIVATE noodle_frontend)
add_test(NAME noodle_smoke_test COMMAND noodle_smoke_test)
add_executable(noodle_interpreter_test tests/interpreter_test.cpp)
target_link_libraries(noodle_interpreter_test PRIVATE noodle_runtime)
add_test(NAME noodle_interpreter_test COMMAND noodle_interpreter_test)
add_executable(noodle_codegen_test tests/codegen_test.cpp)
target_link_libraries(noodle_codegen_test PRIVATE noodle_frontend)
add_test(NAME noodle_codegen_test COMMAND noodle_codegen_test)
